# m1-3 - Render content without JavaScript

## Probe

```bash
# Fetch raw HTML (no JS execution) of homepage and a representative deep page
curl -fsSL -A 'Mozilla/5.0 (compatible; AgentAudit/1.0)' $ORIGIN/ -o /tmp/home.html
wc -c /tmp/home.html

# Strip script/style/comments, count visible text + content-efficiency ratio
python3 -c '
import re,sys
raw=open("/tmp/home.html").read()
h=re.sub(r"(?is)<(script|style|noscript)\b.*?</\1>","",raw)
h=re.sub(r"(?is)<!--.*?-->","",h)
txt=re.sub(r"(?is)<[^>]+>"," ",h)
txt=re.sub(r"\s+"," ",txt).strip()
nchars=len(txt); nbytes=len(raw)
print("text_chars",nchars)
print("html_bytes",nbytes)
print("content_efficiency", round(nchars/nbytes,3) if nbytes else 0)   # visible-text / total-bytes
print("est_tokens", nchars//4)                                          # rough token cost to read the page
print("h1_count", len(re.findall(r"(?is)<h1\b",raw)))
print("anchor_count", len(re.findall(r"(?is)<a\s+[^>]*href=",raw)))
# empty-shell tells: a near-empty root mount that JS later fills
print("shell_markers", bool(re.search(r"(?is)id=[\"\x27](root|app|__next|__nuxt)[\"\x27]\s*>\s*</",raw)) or "enable javascript" in raw.lower())
# alt-text coverage: multimodal agents read alt as the primary image signal
imgs=re.findall(r"(?is)<img\b[^>]*>",raw)
with_alt=sum(1 for t in imgs if re.search(r"(?i)\balt\s*=",t))
print("img_total",len(imgs),"img_with_alt",with_alt,"alt_ratio", round(with_alt/len(imgs),2) if imgs else 1.0)
# document <head> completeness: AI systems lean on these to resolve/disambiguate pages
print("has_canonical", bool(re.search(r"(?is)<link[^>]+rel=[\"\x27]?canonical",raw)))
print("has_lang", bool(re.search(r"(?is)<html[^>]+\blang=",raw)))
print("has_og", bool(re.search(r"(?is)<meta[^>]+property=[\"\x27]?og:(title|description|image)",raw)))
'

# Accessibility tree = the agent's control surface. An agent that falls back to driving the
# UI targets elements by role and name, exactly as a screen reader does: a clickable <div>
# with no accessible name is not reachable, however obvious it looks to a sighted user.
python3 -c '
import re
raw=open("/tmp/home.html").read()
h=re.sub(r"(?is)<(script|style|noscript)\b.*?</\1>","",raw)
# landmarks + heading order
print("has_main", bool(re.search(r"(?is)<main\b|role=[\"\x27]main[\"\x27]",h)))
print("has_nav", bool(re.search(r"(?is)<nav\b|role=[\"\x27]navigation[\"\x27]",h)))
levels=[int(m) for m in re.findall(r"(?is)<h([1-6])\b",h)]
print("heading_skips", sum(1 for a,b in zip(levels,levels[1:]) if b-a>1))
# native controls vs clickable divs/spans
buttons=len(re.findall(r"(?is)<button\b",h))
links=len(re.findall(r"(?is)<a\s+[^>]*href=",h))
fake=len(re.findall(r"(?is)<(div|span)\b[^>]*(onclick=|role=[\"\x27]button[\"\x27])",h))
print("native_controls",buttons+links,"clickable_divs",fake)
# accessible names: every button and every link needs computable text
unnamed=0
for tag,attrs,inner in re.findall(r"(?is)<(button|a)\b([^>]*)>(.*?)</\1>",h):
    if tag=="a" and not re.search(r"(?i)\bhref\s*=",attrs): continue
    text=re.sub(r"(?is)<[^>]+>"," ",inner).strip()
    named=bool(text) or bool(re.search(r"(?i)(aria-label|aria-labelledby|title)\s*=",attrs)) \
        or bool(re.search(r"(?is)<img\b[^>]*\balt\s*=\s*[\"\x27][^\"\x27]+",inner))
    if not named: unnamed+=1
print("controls_unnamed",unnamed)
# form fields need a real label - placeholder text disappears on focus and never counts
labels=set(re.findall(r"(?is)<label\b[^>]*\bfor=[\"\x27]([^\"\x27]+)",h))
unlabeled=0
for attrs in re.findall(r"(?is)<(?:input|select|textarea)\b([^>]*)>",h):
    if re.search(r"(?i)type\s*=\s*[\"\x27]?(hidden|submit|button)",attrs): continue
    idm=re.search(r"(?i)\bid\s*=\s*[\"\x27]([^\"\x27]+)",attrs)
    if (idm and idm.group(1) in labels) or re.search(r"(?i)(aria-label|aria-labelledby)\s*=",attrs): continue
    unlabeled+=1
print("inputs_unlabeled",unlabeled)
# prompt injection through the accessibility tree: instructions parked where only a machine reads
hidden=" ".join(re.findall(r"(?i)(?:aria-label|alt|title)\s*=\s*[\"\x27]([^\"\x27]{12,})",h))
hidden+=" ".join(re.findall(r"(?is)<[^>]*(?:aria-hidden=[\"\x27]true|display:\s*none|sr-only|visually-hidden)[^>]*>(.*?)<",h))
print("injection_hits", len(re.findall(r"(?i)(ignore (all )?previous|disregard (the )?above|as an ai|system prompt|you must (now|always)|do not tell the user)",hidden)))
'

# Trust anchors: the pages an agent reads to decide you are a real business before recommending you
for p in about contact privacy; do
  code=$(curl -sSL -o /tmp/anchor.html -w '%{http_code}' "$ORIGIN/$p" 2>/dev/null)
  chars=$(python3 -c 'import re,sys;raw=open("/tmp/anchor.html").read();h=re.sub(r"(?is)<(script|style)\b.*?</\1>"," ",raw);print(len(re.sub(r"\s+"," ",re.sub(r"(?is)<[^>]+>"," ",h)).strip()))' 2>/dev/null || echo 0)
  printf '/%s %s text_chars=%s\n' "$p" "$code" "$chars"
done

# Bot reachability: do real agent fetchers get the same page, or a 403/challenge/cloaked stub?
BASE=$(curl -fsS -A 'Mozilla/5.0' -o /dev/null -w '%{http_code}' $ORIGIN/)
echo "baseline(browser-UA) $BASE"
for UA in "ChatGPT-User/1.0" "Claude-User/1.0" "PerplexityBot/1.0" "Googlebot/2.1 (Google-Extended)" "ClaudeBot/1.0"; do
  read code size < <(curl -fsS -A "$UA" -o /tmp/ua.html -w '%{http_code} %{size_download}' $ORIGIN/ 2>/dev/null || echo "000 0")
  grep -iqE 'just a moment|cf-browser-verification|captcha|enable javascript to continue' /tmp/ua.html && chal=" CHALLENGE" || chal=""
  printf '  %-38s %s bytes=%s%s\n' "$UA" "$code" "$size" "$chal"
done

# Probe one product/content page too - find a link from sitemap
DEEP=$(curl -fsS $ORIGIN/sitemap.xml | grep -oE '<loc>[^<]+</loc>' | head -2 | tail -1 | sed -E 's|<[^>]+>||g')
[ -n "$DEEP" ] && curl -fsSL "$DEEP" -o /tmp/deep.html && wc -c /tmp/deep.html

# Navigation links in server-rendered HTML
python3 -c '
import re
raw=open("/tmp/home.html").read()
nav=re.findall(r"(?is)<nav\b.*?</nav>", raw)
nav_text=" ".join(nav) if nav else raw
internal_links=set(re.findall(r"(?i)href=\"(/[^\"]*?)\"", nav_text))
print("nav_internal_links", len(internal_links))
' 2>/dev/null

# Conventional agent paths
for p in docs pricing integrations api; do
  curl -fsSI -o /dev/null -w "/$p %{http_code}\n" "$ORIGIN/$p" 2>/dev/null
done
```

## Rubric

| # | Sub-check | Pass when | Weight |
|---|-----------|-----------|--------|
| 1 | Homepage SSR body ≥ 2 KB visible text | Stripped text ≥ 2000 chars | 1 |
| 2 | Exactly one `<h1>` on homepage | Exactly 1 `<h1>` match | 1 |
| 3 | ≥ 10 real `<a href>` links in raw HTML | Anchor count ≥ 10 | 1 |
| 4 | Deep page SSR body ≥ 1 KB visible text | Stripped text on a sample product/article page ≥ 1000 chars | 1 |
| 5 | No "JS required" or empty `<body>` shell | `shell_markers` false - no `enable javascript` and no empty `#root`/`#app`/`#__next` mount | 1 |
| 6 | Agent fetchers reach the page un-challenged | Every agent UA returns 2xx/3xx (not 403/429/503), no Cloudflare/captcha challenge page, and byte size within 50% of baseline (no cloaking/stub) | 1 |
| 7 | Content efficiency reasonable | `content_efficiency` ≥ 0.05 (visible text is at least 5% of bytes - flags JS-bloated shells that cost agents tokens for little signal) | 1 |
| 8 | Alt text on ≥ 80% of images | `alt_ratio` ≥ 0.8 (or no `<img>` on the page) - multimodal agents read `alt` as the primary image signal | 1 |
| 9 | `<head>` resolves & disambiguates | `has_canonical` **and** `has_lang` **and** `has_og` all true (self-referential `rel=canonical`, `<html lang>`, ≥ 1 `og:title`/`description`/`image`) | 1 |
| 10 | Primary navigation in server-rendered HTML | ≥ 5 distinct internal `<a href>` links in `<nav>` or top-level navigation region - no JS-only menus | 1 |
| 11 | Conventional agent paths resolve | At least 2 of `/docs`, `/pricing`, `/integrations`, `/api` return HTTP 2xx or 3xx (not 404) | 1 |
| 12 | Page is operable through the accessibility tree | `controls_unnamed` = 0 **and** `inputs_unlabeled` = 0 **and** `clickable_divs` = 0 **and** `has_main` true - an agent that falls back to driving the UI targets elements by role and name, so a `<div onclick>` with no accessible name is unreachable no matter how obvious the button looks | 1 |
| 13 | (Bonus) No hidden instructions in the accessibility tree | `injection_hits` = 0 - no agent-directed imperatives parked in `aria-label`, `alt`, `title`, or visually-hidden text, where only a machine reads them | 0 |
| 14 | (Bonus) Trust anchor pages carry real content | `/about`, `/contact`, and `/privacy` each return 200 with ≥ 500 chars of visible text - the pages an agent checks before recommending you | 0 |

Status: **Pass** ≥ 10/12 · **Partial** 4-9/12 · **Fail** 0-3/12.

> Sub-check 6 is the one a static-HTML audit usually misses: a site can be perfectly server-rendered yet sit behind bot-management that returns `403`/`Just a moment…` to `ChatGPT-User` or `PerplexityBot`. To an agent that is indistinguishable from "no content." A large byte-size delta between an agent UA and a browser UA is a cloaking signal - surface it verbatim.

## Codebase hints

- **Next.js**: prefer `app/` Server Components or `getServerSideProps`/`getStaticProps` in `pages/`. Avoid `dynamic(() => import(...), { ssr: false })` for above-the-fold copy.
- **React SPA (Vite/CRA)**: migrate to Next.js or add prerender (`react-snap`, `prerender.io`, `vite-plugin-prerender`).
- **Vue SPA**: switch to Nuxt SSR/SSG or use `vite-plugin-vue-ssr-prerender`.
- **PHP (classic / WordPress / Laravel)**: already SSR by default - make sure no critical content is injected by JS after page load.
- **Rails / Django / Phoenix**: SSR by default; audit any `data-react-component` or Stimulus shells that replace content.
- **Static (Hugo/Jekyll/Astro/11ty)**: SSR by default - check that `client:only` islands (Astro) aren't hiding key copy.
- **Bot challenges (sub-check 6)**: if agent UAs get `403`/`Just a moment…`, allowlist them in your WAF. **Cloudflare**: Bot Management → Verified Bots → enable the **Agent** and **Search** categories (post Jul 1 2025, verified bots require per-category enablement; from Sep 15 2025 new zones default-block Agent bots on ad-bearing pages). **Akamai/Fastly**: add the fetcher UAs/ASNs to an allow rule. Prefer Web Bot Auth ([m3-2](./m3-2.md)) over UA allowlists where supported.
- **Accessibility tree (sub-check 12)**: ship real `<button>`/`<a href>` elements instead of `<div onclick>`, give every icon-only control an `aria-label`, and bind each form field to a `<label for>` (a `placeholder` is not a label). The same markup that serves screen readers is what an agent drives when it falls back to the UI.
- **Content efficiency (sub-check 7)**: trim render-blocking inline JSON state, lazy-load below-the-fold, or serve a markdown twin ([m2-3](./m2-3.md)) so agents pay fewer tokens to read you.
